void insertionsort(int a[],int n)
 {
 int i,j,k,key;
 for(i=1;i<n;i++)
 {
 key=a[i];
 j=i-1;
 //Move elements greater then key to one position ahead of their current pos
 while(j>=0 &&a[j]>key)
 {
 a[j+1]=a[j];
 j=j-1;
 }
 a[j+1]=key;
 printf("\nLoop %d:",i);
 for(k=0;k<n;k++)
 {
 printf(" %4d",a[k]);}
 }}
