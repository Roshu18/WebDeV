 int binarysearch(int a[],int low,int high,int data)
 {
 int mid;
while (low<=high)
 {
 mid=(low+high)/2;
 if(a[mid]==data)
 {
 return mid;
 }
 if(a[mid]<data)
 {
 low=mid+1;
 }
 else
 {
 high=mid-1;
 }}
 return-1;
 }
