 #include <stdio.h>
#include <conio.h>
 #define MAX_SIZE 100
 void merge(int arr[],int st,int mid,int end)
 {
 int leftarr[MAX_SIZE],rightarr[MAX_SIZE];
 int n1=(mid-st)+1;
 int n2=end-mid;
 //create temporary array
 //int leftarr[n1], rightarr[n2];
 int i,j,k;
 //copy data to temporary arrays
 for(i=0;i<n1;i++)
 {
 leftarr[i]=arr[st+i];
 }
 for(j=0;j<n2;j++)
 {
 rightarr[j]=arr[mid+1+j];
 }
 //merge the temp array into arr
 i=0;// initial index of left sub array
 j=0;// initial index of right sub array
 k=st; // initial index of merged subarray
 while(i<n1 && j<n2)
 {
 if (leftarr[i]<=rightarr[j])
 {
 arr[k]=leftarr[i];
 i++;
 }
 else
 {
 arr[k]=rightarr[j];
 j++;
 }
 k++;
 }
 //copy the remaining elements of leftarr[]
 while(i<n1)
 {
 arr[k]=leftarr[i];
 i++;
 k++;
 }
//copy the remaining elements of rightarr[]
 while (j<n2)
 {
 arr[k]=rightarr[j];
 j++;
 k++;
 }
 }
 // the subarray to be sorted is in the index range [left- right] void
 mergesort(int arr[],int st,int end)
 {
 if(st<end)
 {
 //calculate mipoint
 int mid=st+(end-st)/2;
 //sort first and second halves
 mergesort(arr,st,mid);
 mergesort(arr,mid+1,end);
 //merge the sorted halves
 merge(arr,st,mid,end);
 }
 }
